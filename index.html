-- Agregar columnas a grupos existente ALTER TABLE grupos ADD COLUMN IF NOT EXISTS es_publica boolean DEFAULT false; ALTER TABLE grupos ADD COLUMN IF NOT EXISTS emoji text DEFAULT '👥'; ALTER TABLE grupos ADD COLUMN IF NOT EXISTS descripcion text; -- Política para que cualquiera pueda ver grupos públicos DROP POLICY IF EXISTS "grupos_select" ON grupos; CREATE POLICY "grupos_select" ON grupos FOR SELECT USING ( es_publica = true OR creado_por = auth.uid() OR id IN (SELECT grupo_id FROM grupo_miembros WHERE user_id = auth.uid()) ); -- Política para que cualquiera pueda unirse a grupos públicos DROP POLICY IF EXISTS "miembros_insert" ON grupo_miembros; CREATE POLICY "miembros_insert" ON grupo_miembros FOR INSERT WITH CHECK ( auth.uid() = user_id AND ( EXISTS (SELECT 1 FROM grupos WHERE id = grupo_id AND es_publica = true) OR EXISTS (SELECT 1 FROM grupos WHERE id = grupo_id AND creado_por = auth.uid()) ) );
